component MultiLap
{
_currentLap = 0;
lapCount = 3;
Refill = true;
PlaySound = true;

function OnCollisionEnter(other)
{
if (other.Type == "Human" && other.IsMine)
{
if (self.Refill)
{
other.RefillImmediate();
}
if (self.PlaySound)
{
other.PlaySound("Checkpoint");
if (self._currentLap < self.lapCount)
{
self._currentLap = self._currentLap + 1;
Game.Print(other.Player.Name +" " +"Lap: " + self._currentLap + "/" + self.lapCount);
}
}
other.Player.SpawnPoint = self.MapObject.Position;
}
}
function OnSecond()
{
if (self._currentLap >= self.lapCount)
{
finish = Map.FindMapObjectsByName("Finish");
for (finish_ in finish)
{
finish_.Active = true;
}
}
}
}
