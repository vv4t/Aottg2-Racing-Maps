	function OnPlayerLeave(player)
	{
        return;
        if (player.ID == Network.MyPlayer.ID)
        {
            if (self._isPassenger)
            {
                self._isPassenger = false;
                self._hasPassenger = false;
                self.NetworkView.SendMessage(player, "p" + Convert.ToString(self._hasPassenger)); 
            }
            elif(self.NetworkView.Owner == Network.MyPlayer || Network.IsMasterClient)
            {
                self._inUse = false;
                self.NetworkView.Transfer(Network.MasterClient);
            }
        }
	}

	function OnCharacterDie(victim, killer, killerName)
	{
        if (victim.IsMine)
        {
            Camera.SetManual(false);
            Map.DestroyMapObject(self.MapObject, true);
        }

        return;

        if (victim.Type != "Human") { return; }

        if (victim.Player.ID == Network.MyPlayer.ID)
        {
            if (self._isPassenger)
            {
                self._isPassenger = false;
                self._hasPassenger = false;
                self.NetworkView.SendMessage(victim.Player, "p" + Convert.ToString(self._hasPassenger)); 
            }
            elif(self.NetworkView.Owner == Network.MyPlayer || Network.IsMasterClient)
            {
                self._inUse = false;
			    self.NetworkView.Transfer(Network.MasterClient);
                Network.MyPlayer.Character.Transform.GetTransform(HumanTransformEnum.Armature).SetRenderersEnabled(true);
            }
        }
	}

    function OnNetworkStream()
    {
        self._inUse = self.NetworkView.ReceiveStream();
        self._soundEngine = self.NetworkView.ReceiveStream();
        self._soundVelocity = self.NetworkView.ReceiveStream();
    }

    function SendNetworkStream()
    {
        self.NetworkView.SendStream(self._inUse);
        self.NetworkView.SendStream(self._soundEngine);
        self.NetworkView.SendStream(self._soundVelocity);
    }

    function OnNetworkMessage(sender, message)
    {
        if (Network.IsMasterClient && self.NetworkView.Owner == Network.MyPlayer && !self._inUse && message == "request")
        {
            self.NetworkView.Transfer(sender);
        }
        elif (String.StartsWith(message, "p"))
        {
            self._hasPassenger = Convert.ToBool(String.Substring(message, 1));
        }
        elif(message == "enter") {
            self._doorSlamAudio.Active = true;
            return;
        }
    }

    function OnNetworkTransfer(oldOwner, newOwner)
    {
        if (newOwner == Network.MasterClient && Network.IsMasterClient)
        {
            self._inUse = false;
        }
        elif (newOwner == Network.MyPlayer)
        {
            human = self._collidingHuman;
            if (human != null && human.Type == "Human")
            {
                self.OnEnterCarRider();
            }
            else
            {
                self.NetworkView.Transfer(Network.MasterClient);
            }
        }
        if(Network.IsMasterClient && newOwner != Network.MyPlayer) {
            Game.Print(newOwner.ID + "] Name: " + newOwner.Name);
            self._human = newOwner.Character;
        }
    }

    function OnCollisionStay(other)
    {
        if (other.Type != "Human" || !other.IsMine) { return; }
        
        vector = other.Position - self.MapObject.Position;
        project = Vector3.Project(vector, self.MapObject.Right);
        if (project.Normalized == self.MapObject.Right && !self._inUse && !other.IsMounted)
        {
            UI.SetLabelForTime("MiddleCenter", "Press " + Input.GetKeyName("Interaction/Interact") + " to drive", 0.1);
            self._collidingHuman = other;
            self._collidingDrive = true;
        }
        elif (project.Normalized != self.MapObject.Right && !self._hasPassenger && !other.IsMounted)
        {
            UI.SetLabelForTime("MiddleCenter", "Press " + Input.GetKeyName("Interaction/Interact") + " to get in", 0.1);
            self._collidingHuman = other;
            self._collidingPassenger = true;
        }
    }

    function OnTick()
    {
        self._collidingHuman = null;
        self._collidingDrive = false;
        self._collidingPassenger = false;

        vector = self._rigidbody.GetVelocity();
        y = vector.Y;
        vector.Y = 0;
        velocity = vector.Magnitude;

        self.PlaySounds(self._soundEngine, self._soundVelocity);

        if(self._leftBackWheel != null) {
            rotationSpeed = Math.Clamp(velocity, 0, 10);
            currentMax = self.MaxSteeringAngle;

            if (self._isDrifting) { currentMax = self.DriftMaxSteeringAngle; }
            yRotation = Math.Clamp(self._drivingAngle / (currentMax / 45), -45, 45);
            xRotation = Math.Sign(self._enginePower) * rotationSpeed;

            self._leftFrontWheel.LocalRotation = Vector3(self._leftFrontWheel.LocalRotation.X + xRotation, yRotation, self._leftFrontWheel.LocalRotation.Z);
            self._rightFrontWheel.LocalRotation = Vector3(self._rightFrontWheel.LocalRotation.X + xRotation, yRotation, self._rightFrontWheel.LocalRotation.Z);
            self._leftBackWheel.Rotate(Vector3(xRotation, 0, 0));
            self._rightBackWheel.Rotate(Vector3(xRotation, 0, 0));
        }
        if(self.NetworkView.Owner != Network.MyPlayer) { return; } 

        if(self._inUse && self._human != null) {
            speedDrop = self._previousSpeed - velocity;
            if(speedDrop >= self.CrashDecelerationThreshold) { self.OnCrash(speedDrop); }
            self._previousSpeed = velocity;

            UI.SetLabel(UILabelEnum.MiddleRight, "Speed[" + String.FormatFloat(velocity, 2) + "]");
            Network.MyPlayer.Character.Rigidbody.Velocity = vector;
            self._soundEngine = self._enginePower;
            self._soundVelocity = velocity;
            self.CheckIsGrounded();

            project = Vector3.Project(vector, self.MapObject.Forward);
            self._isBraking = Input.GetKeyHold(SettingsManager.ReturnKeyString("Brake"));
            
            if(self._driftTimer > 0.0) {
                self._driftTimer -= Time.TickTime;
                self._isDrifting = true;
            } else {
                self._driftTimer = 0.0;
                self._isDrifting = false;
            }

            if(self._isBraking) { self._currentBrakingMultiplier = Math.Lerp(self._currentBrakingMultiplier, self.BrakingSteeringScale, self.BrakingMultiplierLerpScale); }
            else { self._currentBrakingMultiplier = Math.Lerp(self._currentBrakingMultiplier, 1.0, self.BrakingMultiplierLerpScale); }

            forwardMultiplier = 1.0;
            if(project.Normalized != self.MapObject.Forward && !self._isDrifting) { forwardMultiplier = -1.0; }

            rotatePower = 1.0;
            if(velocity < 0.001) { rotatePower = 0; }
            elif(velocity < self._lowVelocityThreshold) { rotatePower = velocity / self._lowVelocityThreshold; }
            
            rotatePower *= forwardMultiplier;
            if(self._isBraking) { rotatePower *= self._currentBrakingMultiplier; }
            if(self._isDrifting) { rotatePower *= self.DriftSteeringScale; }

            if(Input.GetKeyHold("General/Back")) { self._enginePower = Math.Lerp(self._enginePower, self.MaxEngineSpeed * -1, self.EngineSpeed); }
            elif(Input.GetKeyHold(SettingsManager.ReturnKeyString("Accelerate"))) { self._enginePower = Math.Lerp(self._enginePower, self.MaxEngineSpeed, self.EngineSpeed); }
            else{ self._enginePower = Math.Lerp(self._enginePower, 0, 0.05); }

            currentSteeringSpeed = self.SteeringSpeed;
            currentMaxAngle = self.MaxSteeringAngle;

            if(self._isDrifting) {
                self._driftHoldTime = Math.Lerp(self._driftHoldTime, 1.0, self.DriftSteeringRampUp);
                currentSteeringSpeed = Math.Lerp(self.DriftSteeringSpeedMin, self.DriftSteeringSpeedMax, self._driftHoldTime);
                currentSteeringSpeed = Math.Clamp(currentSteeringSpeed, self.DriftSteeringSpeedMin, self.DriftSteeringSpeedMax);
                currentMaxAngle = self.DriftMaxSteeringAngle;
            } else {
                self._driftHoldTime = Math.Lerp(self._driftHoldTime, 0.0, self.DriftSteeringRampDown);
            }
            if (Input.GetKeyHold("General/Left")) { self._drivingAngle = Math.Lerp(self._drivingAngle, (-1.0 * currentMaxAngle), currentSteeringSpeed); }
            elif (Input.GetKeyHold("General/Right")) { self._drivingAngle = Math.Lerp(self._drivingAngle, currentMaxAngle, currentSteeringSpeed); }
            else { self._drivingAngle = Math.Lerp(self._drivingAngle, 0, 0.1); }

            if(self._steeringWheel != null) {
                visualWheelTurn = self._drivingAngle * 2.5;
                tiltX = self._steeringWheel.LocalRotation.X;
                tiltY = self._steeringWheel.LocalRotation.Y;

                self._steeringWheel.LocalRotation = Vector3(tiltX, tiltY, visualWheelTurn);
            }

            if(self._isGrounded) {
                self.MapObject.Transform.Rotate(Vector3(0, self._drivingAngle * Time.TickTime * rotatePower, 0));
                forceDirection = self.MapObject.Forward;
                engineMultiplier = 1.0;
                if(self._isDrifting && velocity > 2.0)  {
                    self._driftAngle = Vector3.Angle(vector.Normalized, self.MapObject.Forward);

                    speedFactor = Math.Clamp(velocity / self.MaxMoveSpeed, 0.3, 1.0);
                    blendAmount = 0.4 + (speedFactor * 0.3);
                    forceDirection = Vector3.Lerp(self.MapObject.Forward, vector.Normalized, blendAmount);
                    engineMultiplier = 0.75 + (speedFactor * 0.15);

                    if(self._driftAngle > 25.0) {
                        counterSteerAmount = Math.Clamp((self._driftAngle - 25.0) / 65.0, 0.0, 1.0);
                        cross = Vector3.Cross(self.MapObject.Forward, vector.Normalized);
                        counterDirection = Math.Sign(cross.Y) * -1.0;
                        assistAngle = counterDirection * currentMaxAngle * self.DriftCounterSteerAssist * counterSteerAmount;
                        self._drivingAngle = Math.Lerp(self._drivingAngle, self._drivingAngle + assistAngle, 0.3);
                    }
                }

                self._rigidbody.AddForce(forceDirection * (self.MoveForce * engineMultiplier) * (self._enginePower / self.MaxEngineSpeed));
                if(Math.Abs(self._enginePower) < 1) { self.SlowToStop(); }
                if(self._isDrifting) { self._driftTransition = Math.Lerp(self._driftTransition, 1.0, 0.15); }
                else { self._driftTransition = Math.Lerp(self._driftTransition, 0.0, 0.05); }

                baseGrip = self.VelocityFacing - (self._velocityFacingDiff * velocity / self.MaxMoveSpeed);
                targetGrip = Math.Lerp(baseGrip, self.DriftGrip, self._driftTransition);

                if(self._isBraking && !self._isDrifting) { targetGrip *= self.BrakingVelocityFacingScale * (self._currentBrakingMultiplier / 1.0); }
                self._currentGrip = Math.Lerp(self._currentGrip, targetGrip, 0.02);

                newVector = Vector3.Lerp(vector.Normalized, self.MapObject.Transform.Forward * forwardMultiplier, self._currentGrip) * velocity;

                if(self._isDrifting) {

                    sidewaysFactor = Math.Abs(Vector3.Dot(newVector.Normalized, self.MapObject.Right));
                    speedLossFactor = 1.0 - (self.DriftSpeedLoss * sidewaysFactor);

                    newVector = newVector.Magnitude * speedLossFactor * newVector.Normalized;
                } 
                elif(self._isBraking) { newVector = newVector.Magnitude * (1.0 - (self.BrakingScale * self._currentBrakingMultiplier / 1.0)) * newVector.Normalized; }
                newVector.Y = y;

                self._rigidbody.SetVelocity(newVector);
                self.LimitSpeed();
                self.FaceSurface();
                
                if(self._wasDriftingLastFrame && !self._isDrifting && velocity > 5.0) {
                    currentVel = self._rigidbody.GetVelocity();
                    currentVel.Y = 0;
                    if(currentVel.Magnitude > 0.1) {
                        boostedVel = currentVel.Normalized * Math.Min(currentVel.Magnitude * self.DriftExitBoost, self.MaxMoveSpeed);
                        boostedVel.Y = y;
                        self._rigidbody.SetVelocity(boostedVel);
                    }
                }
                self._wasDriftingLastFrame = self._isDrifting;

                if(self._leftBackWheelSmoke != null) {
                    if(self._isDrifting && velocity > 30.0) {
                        smokeIntensity = Math.Clamp(self._driftAngle / 35.0, 0.5, 1.5);
                        self._leftBackWheelSmoke.Scale = Vector3(0.5 * smokeIntensity);
                        self._rightBackWheelSmoke.Scale = Vector3(0.5 * smokeIntensity);
                        self._leftBackWheelSmoke.Active = true;
                        self._rightBackWheelSmoke.Active = true;
                        self._driftingSFX.Active = true;
                    } 
                    else {
                        self._leftBackWheelSmoke.Active = false;
                        self._rightBackWheelSmoke.Active = false;
                        self._driftingSFX.Active = false;
                    }
                }
            }
        }
    }

    function OnCrash(damage)
    {   
        me = Network.MyPlayer.Character;
        self.Health -= damage;
        if(self.Health <= 0) { 
            self._explosion.Scale = Vector3((damage / 10) * 0.5);
            self._explosion.Active = true;
            self.Health = 0;
        }
        if(self._driftingSFX.Active) { self._driftingSFX.Active = false; }
        #Game.Print(damage + " Health: " + self.Health);
        
        Game.SpawnEffect(EffectNameEnum.Boom5, self._leftFrontWheel.Position, Vector3(), (damage / 10) * 0.5);

        if(damage >= self.Health) {
            if(me != null) {
                self.OnExitCarRider();
                carVelocity = self._rigidbody.GetVelocity();
                if(carVelocity.Magnitude > 0) { me.Velocity = carVelocity * (damage / 20); }
            }
        }
    }

    function OnFrame()
    {   
        if (self._human != null) 
        {
            if(self._human != Network.MyPlayer.Character) { return; }
            self._human.PlayAnimation("Armature|horse_idle"); 
            changeCamKey = Input.GetKeyDown(SettingsManager.ReturnKeyString("Change Camera"));
            toggleCamLockKey = Input.GetKeyDown(SettingsManager.ReturnKeyString("Toggle Camera Lock"));
            if (changeCamKey) {
                if(self._isCameraLocked) {
                    if(!Camera.IsManual) { Camera.SetManual(true); }
                    self._currentCamera = Math.Mod(self._currentCamera + 1, 4); 
                    if(self._currentCamera == 0) { Network.MyPlayer.Character.Transform.GetTransform(HumanTransformEnum.Armature).SetRenderersEnabled(false); }
                    else { Network.MyPlayer.Character.Transform.GetTransform(HumanTransformEnum.Armature).SetRenderersEnabled(true); }
                    Game.Print("Camera: " + self._cameraNames.Get(self._currentCamera)); 
                }
                else { self.ToggleCameraMode(); }
            }
            elif(toggleCamLockKey) {
                self.ToggleCameraLock();
            }
        }
        
        if (self.NetworkView.Owner == Network.MyPlayer && self._inUse)
        {
            self._human = self.NetworkView.Owner.Character;
            if (Input.GetKeyDown("Interaction/Interact"))
            {
                self.OnExitCarRider();
                return;
            }
            flipCarKey = Input.GetKeyDown(SettingsManager.ReturnKeyString("Flip Car"));
            driftKey = Input.GetKeyHold(SettingsManager.ReturnKeyString("Drift"));
            if (flipCarKey)
            {   
                self._rigidbody.SetVelocity(Vector3.Zero);
                self.MapObject.Transform.Position = self.MapObject.Transform.Position + Vector3.Up * 3.0;
                self.MapObject.Rotation = Vector3(0,Camera.Rotation.Y, 0);
            }
            if(driftKey) { 
                if(self._isDrifting) { return; }
                speedFraction = Math.Clamp(self._currentVelocity / self.MaxMoveSpeed, 0.0, 1.0);
                self._driftTimer = Math.Lerp(self.DriftDurationMin, self.DriftDurationMax, speedFraction);
            } 

            self._isBraking = Input.GetKeyHold(SettingsManager.ReturnKeyString("Brake"));
            
            if (self.NetworkView.Owner.Character == null || self.NetworkView.Owner.Character.Type != "Human" || self.NetworkView.Owner.Character.MountedTransform != self._transform)
            {
                return;
                self.NetworkView.Transfer(Network.MasterClient);
                self._inUse = false;
                return;
            }
        }
        elif (self._hasPassenger && self._human != null)
        {
            if (self._human == null || self._human.MountedMapObject != self.MapObject)
            {
                self._hasPassenger = false;
                return;
            }
            if (Input.GetKeyDown("Interaction/Interact"))
            {
                self._human.Unmount();
                self._human = null;
                self.MapObject.Position += self.MapObject.Right * 1.0;
                Camera.SetManual(false);
                self._hasPassenger = false;
                self._isPassenger = false;
                self.NetworkView.SendMessageOthers("p" + Convert.ToString(self._hasPassenger));
            }
        }
        if (self._collidingHuman != null && !self._collidingHuman.IsMounted && Input.GetKeyDown("Interaction/Interact"))
        {
            if (self._collidingDrive && !self._inUse)
            {
                if (self.NetworkView.Owner == Network.MyPlayer) 
                {
                    self.OnEnterCarRider();
                }
                else
                {
                    self.NetworkView.SendMessage(self.NetworkView.Owner, "request");
                }
            }
            elif (self._collidingPassenger && !self._hasPassenger)
            {
                self._human = self._collidingHuman;
                self._human.MountTransform(self._transform, Vector3(-0.3, 0.6, -0.1), Vector3.Zero);
                self._human.ClearHooks();
                self._hasPassenger = true;
                self._isPassenger = true;
                self.NetworkView.SendMessageOthers("p" + Convert.ToString(self._hasPassenger));
            }
        }
    }

    
    function OnEnterCarRider()
    {
        if(self.Health <= 0) { return; }
        self._human = self._collidingHuman;
        self._human.MountTransform(self._transform, self.MountPositionOffsetRider, Vector3.Zero);
        self._human.ClearHooks();
        self._inUse = true;
        if(self._isCameraLocked) { Camera.SetManual(true); }
        Input.SetKeyDefaultEnabled(InputGeneralEnum.ChangeCamera, false);
        
    }

    function OnExitCarRider()
    {
        self._human.Unmount();
        self._human = null;
        self.MapObject.Position += self.MapObject.Right * -1.0;
        
        # self.NetworkView.Transfer(Network.MasterClient);
        self._inUse = false;
        if(Camera.IsManual) { Camera.SetManual(false); }
        Input.SetKeyDefaultEnabled(InputGeneralEnum.ChangeCamera, true);
        Network.MyPlayer.Character.Transform.GetTransform(HumanTransformEnum.Armature).SetRenderersEnabled(true);
    }


    function ToggleCameraLock()
    {
        self._isCameraLocked = !self._isCameraLocked;
        Camera.SetManual(self._isCameraLocked);
    }

    function ToggleCameraMode()
    {   
        if(Camera.IsManual) { Camera.SetManual(false); }
        camMode = Camera.CameraMode;
        if(camMode == CameraModeEnum.TPS) { Camera.SetCameraMode(CameraModeEnum.Original); Network.MyPlayer.Character.Transform.GetTransform(HumanTransformEnum.Armature).SetRenderersEnabled(true); }
        elif(camMode == CameraModeEnum.Original) { Camera.SetCameraMode(CameraModeEnum.FPS); Network.MyPlayer.Character.Transform.GetTransform(HumanTransformEnum.Armature).SetRenderersEnabled(false); }
        else { Camera.SetCameraMode(CameraModeEnum.TPS); Network.MyPlayer.Character.Transform.GetTransform(HumanTransformEnum.Armature).SetRenderersEnabled(true); }
        Game.Print("Camera Mode: " + Camera.CameraMode);
    }

        function OnLateFrame()
        {   
            me = Network.MyPlayer.Character;
            if(self._human != null && self._human == me) 
            {   
                currentCamera = self._currentCamera;
                mapObjectPos = self.MapObject.Position;
                lookAtOffset = self.CameraLookAtOffset;

                velocityDir = self._rigidbody.GetVelocity();
                velocityDir.Y = 0;
                
                if (velocityDir.Magnitude < 1.0) { velocityDir = self.MapObject.Forward; }
                else { velocityDir = velocityDir.Normalized; }

                camForward = Vector3.Lerp(self.MapObject.Forward, velocityDir, 0.6).Normalized;

                if (currentCamera == 0)
                {   
                    firstPersonCameraOffset = self.FirstPersonCameraOffset;
                    cameraSetPosition = self._human.Position + self.MapObject.Up * firstPersonCameraOffset.Y + self.MapObject.Forward * firstPersonCameraOffset.X;
                    lookAt = mapObjectPos + self.MapObject.Forward * lookAtOffset.X + self.MapObject.Up * lookAtOffset.Y;
                }
                elif (currentCamera == 1)
                {   
                    bumperCameraOffset = self.BumperViewCameraOffset;
                    cameraSetPosition = mapObjectPos + self.MapObject.Up * bumperCameraOffset.Y + self.MapObject.Forward * bumperCameraOffset.X;
                    lookAt = mapObjectPos + self.MapObject.Forward * lookAtOffset.X + self.MapObject.Up * lookAtOffset.Y;
                }
                elif (currentCamera == 2)
                {   
                    thirdPersonCameraOffset = self.ThirdPersonCameraOffset;
                    cameraSetPosition = mapObjectPos + self.MapObject.Up * thirdPersonCameraOffset.Y + camForward * thirdPersonCameraOffset.X;
                    lookAt = mapObjectPos + camForward * lookAtOffset.X + self.MapObject.Up * lookAtOffset.Y;
                }
                elif (currentCamera == 3)
                {   
                    thirdPersonExtendedCameraOffset = self.ThirdPersonExtendedCameraOffset;
                    cameraSetPosition = mapObjectPos + self.MapObject.Up * thirdPersonExtendedCameraOffset.Y + camForward * thirdPersonExtendedCameraOffset.X;
                    lookAt = mapObjectPos + camForward * lookAtOffset.X + self.MapObject.Up * lookAtOffset.Y; 
                }

                if(self._isCameraLocked) {
                    Camera.SetPosition(cameraSetPosition); 
                    Camera.LookAt(lookAt);
                }
            }
        }

        function SlowToStop()
        {
            velocity = self._rigidbody.GetVelocity();
            velocityY = velocity.Y;
            velocity.Y = 0;

            if (velocity.Magnitude < 1) { self._rigidbody.SetVelocity(Vector3.Up * velocityY); }
            else { self._rigidbody.AddForce(self.SlowDownScale * -1.0 * velocity + velocityY * Vector3.Up); }
        }

        function LimitSpeed()
        {
            velocity = self._rigidbody.GetVelocity();
            velocityY = velocity.Y;
            velocity.Y = 0;
            velocity = velocity.Normalized * Math.Min(velocity.Magnitude, self.MaxMoveSpeed);
            velocity.Y = velocityY;
            self._rigidbody.SetVelocity(velocity);
        }

        function CheckIsGrounded()
        {
            offset = self.MapObject.Up * self.MapObject.Scale.X * -0.5;
            for (wheel in self._wheels)
            {
                hit = Physics.LineCast(wheel.Position, wheel.Position + offset, "MapObjects");
                if (hit != null)
                {
                    self._isGrounded = true;
                    return;
                }
            }
            self._isGrounded = false;
        }

        function FaceSurface()
        {
            hit = Physics.LineCast(self.MapObject.Transform.Position, self.MapObject.Transform.Position + self.MapObject.Transform.Up * -2.0, "MapObjects");
            if (hit == null) { return; }
            difference = hit.Normal - self.MapObject.Transform.Up;
            difference.Y = 0;
            self._rigidbody.AddTorque(difference * self.TorqueForce, ForceModeEnum.Acceleration);
        }

        function PlaySounds(engine, velocity)
        {
            self._ignition.Position = self.MapObject.Position;
            self._exhaust.Position = self.MapObject.Position;
            self._lowRumble.Position = self.MapObject.Position;
            self._highRumble.Position = self.MapObject.Position;

            if (!self._inUse && self._engineOn)
            {
                self._engineOn = false;
                self.StopSounds();
            }
            elif (self._inUse && !self._engineOn)
            {
                self._engineOn = true;
                self.Ignition();
            }

            if (self._engineOn && Math.Abs(engine) < 5) { self._canExhaust = true; }
            elif (self._engineOn && engine > 5 && self._canExhaust) 
            { 
                self._canExhaust = false; 
                self.Exhaust();
            }

            if(velocity > 1 && self._engineOn) { self.HighRumble(); }
            if (Math.Abs(engine) < self.MaxEngineSpeed / 5) { self._highRumble.Active = false; }
        }

        function StopSounds()
        {
            self._ignition.Active = false;
            self._exhaust.Active = false;
            self._lowRumble.Active = false;
            self._highRumble.Active = false;
        }

        coroutine LowRumble()
        {
            if (self._lowRumble.Active) { return; }
            self._lowRumble.Active = true;
            wait 3.0;
            self._lowRumble.Active = false;
        }

        coroutine HighRumble()
        {
            if (self._highRumble.Active) { return; }
            self._highRumble.Active = true;
            wait 12.0;
            self._highRumble.Active = false;
        }

        coroutine Ignition()
        {
            self._ignition.Active = false;
            wait 0.1;
            self._ignition.Active = true;
        }

        coroutine Exhaust()
        {
            if (self._exhaust.Active) { return; }
            self._exhaust.Active = false;
            wait 0.1;
            self._exhaust.Active = true;
            wait 4.7;
            self._exhaust.Active = false;
        }
}

    }

    function GetAndDisableKeybindSlots()
    {
        disabledKeys = self.DisabledBaseKeys;
        if(disabledKeys == "") { return; }
        Game.Print(Utils.ColorStr("Disabled Base Game Keys for Custom Key Slots: ", self.PrimaryUIColor));
        secondaryColor = self.SecondaryUIColor;
        accentColor = self.AccentUIColor;
        keybindID = self._keybindIdentifier;
        settingID = self._settingIdentifier;

        for(prop in String.Split(disabledKeys, ",")) {
            key = self._settings.Get(keybindID + prop);
            keyName = Utils.ColorStr(Input.GetKeyName(key), accentColor);
            
            Game.Print(Utils.ColorStr(prop, self.PrimaryUIColor) + " [" + keyName + "] " + Utils.ColorStr(key, secondaryColor));
            self._disabledPropKeys.Add(key);
            Input.SetKeyDefaultEnabled(key, false); 
        }
    }

    function OnButtonClick(buttonName)
    {
        args = String.Split(buttonName, "|");
        button = args.Get(0);

        if(button == "Key") {
            self._pendingRebind = args.Get(1);
            #Game.Print(self._pendingRebind);
            self.ShowAllBindsPopup(self._pendingRebind);
            return;
        }
        elif(button == "Rebind") {
            pendingRebind = args.Get(1);
            popupName = self._popupName;
            if(self._pendingRebind != pendingRebind) {
                UI.ClearPopup(popupName);
                UI.HidePopup(popupName);
                Game.Print("Error try again");
                return;
            }
            key = args.Get(2);
            previousKey = self._settings.Get(pendingRebind);
            
            if(self._disabledPropKeys.Contains(pendingRebind)) {
                Input.SetKeyDefaultEnabled(key, false);
                Input.SetKeyDefaultEnabled(previousKey, true);
                #Game.Print(previousKey + " Current Key: " + key);
            }
            #Game.Print(key + " | KeyName: " + Input.GetKeyName(key));
            self.SetSetting(pendingRebind, key);
            self.SaveSettings();
            self.ShowKeybindPopup();
            self._pendingRebind = "";
            return;
        }
    }

    function IsKeybind(propString)
    {
        return String.SubstringWithLength(propString, 0, 1) == "K";
    }

    function IsSetting(propString)
    {
        return String.SubstringWithLength(propString, 0, 1) == "S";
    }

        popupName = self._popupName;

    }

    function ShowAllBindsPopup(pendingRebind)
    {
        popupName = self._popupName; 
        UI.ClearPopup(popupName);
        primaryColor = self.PrimaryUIColor;
        secondaryColor = self.SecondaryUIColor;
        accentColor = self.AccentUIColor;
        UI.AddPopupLabel(popupName, Utils.ColorStr("Rebinding the ", primaryColor) + Utils.ColorStr(pendingRebind, accentColor) + Utils.ColorStr(" Key", primaryColor));

        for(key in self._allAvailableKeybinds) {
            if(self._settings.Values.Contains(key)) { continue; }
            customKeyName = String.Split(pendingRebind, "_").Get(1);
            customKeybind = Utils.ColorStr(customKeyName, primaryColor);
            keyName = Utils.ColorStr(Input.GetKeyName(key), accentColor);
            keybindString = Utils.ColorStr(key, secondaryColor);
            
            UI.AddPopupButton(popupName, "Rebind|" + pendingRebind + "|" + key, "Rebind " + customKeybind + " to [" + keyName + "] " + keybindString);
        }
    }

    function AddSavedKeybinds(popupName)
    {
        primaryColor = self.PrimaryUIColor;
        secondaryColor = self.SecondaryUIColor;
        accentColor = self.AccentUIColor;

        for(prop in self._settings.Keys) {
            if(!self.IsKeybind(prop)) { continue; }
            keyName = Utils.ColorStr(Input.GetKeyName(self._settings.Get(prop)), accentColor);
            keybindString = Utils.ColorStr(self._settings.Get(prop), secondaryColor);
            propName = String.Split(prop, "_").Get(1);
            #Game.Print(prop);
            customKey = Utils.ColorStr(propName, primaryColor);

            UI.AddPopupButton(popupName, "Key|" + prop, "Rebind " + customKey + " [" + keyName + "] " + keybindString);
        }
    }

    function SetSetting(propertyName, keyString)
    {
        self._settings.Set(propertyName, keyString);
    }

    }

    function SaveSettings()
    {   
        PData.SaveToFile(self._fileName, "", self._settings, false, false, true);
    }

    function LoadKeybinds()
    {   
        return;
        fileName = self._fileName;
        keybindFileExists = PersistentData.FileExists(fileName);
        if(!keybindFileExists) {
            if(self._settings.Count == 0) {
                Game.Print("Set your Keybinds first in function SetKeybinds()");
                return;
            }
            self.SaveSettings();
            return;
        }

        for(prop in self._settings.Keys) {
            if(!self.IsKeybind(prop)) { continue; }
            savedKeybind = PData.LoadFromFile(fileName, prop, false, false);
            if (savedKeybind == "Interaction/Function4") {
                savedKeybind = "Interaction/QuickSelect1";
            }
            if(savedKeybind != "Empty") {
                self._settings.Set(prop, savedKeybind);
            }
        }
        self.GetAndDisableKeybindSlots();
    }

extension PData
{
    _lastSaveFile = "";
    _lastLoadFile = "";

    function SaveToFile(fileName, propName, value, encrypted, serialized, isMultiple) #If isMultiple value should = Dict
    {   
        if(!PersistentData.IsValidFileName(fileName)) {
            Game.Print("Invalid File Name: " + fileName);
            return;
        }
        if(self._lastSaveFile != fileName) {
            self._lastSaveFile = fileName;
            PersistentData.Clear();
        }

        if(isMultiple) {
            for(prop in value.Keys) { PersistentData.SetProperty(prop, value.Get(prop)); }
            PersistentData.SaveToFile(fileName, encrypted);
            return;
        }
        PersistentData.SetProperty(propName, value);
        PersistentData.SaveToFile(fileName, encrypted);
    }

    function LoadFromFile(fileName, propName, encrypted, serialized)
    {
        if(self._lastLoadFile != fileName) {
            PersistentData.Clear();
            self._lastLoadFile = fileName;
        }

        PersistentData.LoadFromFile(fileName, encrypted);
        data = PersistentData.GetProperty(propName, "Empty");
        if(serialized) { return Json.LoadFromString(data); }
        return data;
    }
}
