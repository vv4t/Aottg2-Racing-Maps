component TextSign
{
    Description = "Puts a label in the middle of the screen when near the object.";

    TextLine1 = "";
    TextLine2 = "";
    TextLine3 = "";
    TextLine4 = "";
    TextLine5 = "";
    TextLine6 = "";
    TextLine7 = "";

    LetterSize = 25;

    HitboxSize = Vector3(5,5,5);

    function Init()
    {
        # preventing the parent scale from affecting the collider size
        x = self.HitboxSize.X/self.MapObject.Scale.X;
        y = self.HitboxSize.Y/self.MapObject.Scale.Y;
        z = self.HitboxSize.Z/self.MapObject.Scale.Z;
        self.HitboxSize = Vector3(x,y,z);

        self.MapObject.AddBoxCollider("Region", "Humans", Vector3(0,0,0), self.HitboxSize);
    }
    
    function OnCollisionStay(other)
    {
        if (other.Type == "Human" && other.IsMine)
        {
            str = self.TextLine1 + String.Newline
                + self.TextLine2 + String.Newline
                + self.TextLine3 + String.Newline
                + self.TextLine4 + String.Newline
                + self.TextLine5 + String.Newline
                + self.TextLine6 + String.Newline
                + self.TextLine7 + String.Newline;
            str = "<size=" + self.LetterSize + ">" + str + "</size>";
            UI.SetLabelForTime("MiddleCenter", str, 0.1);
        }
    }
}

component SignalActive {
  Behaviour = "Disappear";
  BehaviourDropbox = "Disappear,Appear,Toggle,None";
  
  TargetObjectIDs = "";
  TargetObjectIDsTooltip = "The IDs of the objects which should be targeted separated by space. e.g. 4 5 6";
  
  NetworkSync = false;
  NetworkSyncTooltip = "Enable this if visibility should be synced across clients. Networked also has to be enabled.";
  
  TriggerLimit = 0;
  TriggerLimitTooltip = "The maximum times the component can be triggered. No limit if set to 0.";
  
  _targetObjects = List();
  _internalState = "";
  _triggerCount = 0;
  
  function OnGameStart() {
    for (id in String.Split(self.TargetObjectIDs, " ")) {
      obj = Map.FindMapObjectByID(Convert.ToInt(id));
      self._targetObjects.Add(obj);
    }
    self.SyncState();
  }
  
  function OnNetworkMessage(sender, message) {
    data = Json.LoadFromString(message);
    self._internalState = data.Get("internalState");
    self._triggerCount = data.Get("triggerCount");
    self.SyncObject();
  }
  
  function OnPlayerJoin(player) {
    if (self.NetworkSync && self.NetworkView.Owner == Network.MyPlayer) {
      self.NetworkView.SendMessage(player, self.CreateSyncMessage());
    }
  }
  
  function OnCollisionEnter(other) {
    if (other.Type == "Human" && other.IsMine) {
      if (self.TriggerLimit != 0 && self._triggerCount >= self.TriggerLimit) {
        return;
      }
      
      self._triggerCount += 1;
      
      for (obj in self._targetObjects) {
        if (self.Behaviour == "Disappear") {
          obj.Active = false;
        } elif (self.Behaviour == "Appear") {
          obj.Active = true;
        } elif (self.Behaviour == "Toggle") {
          obj.Active = !obj.Active;
        }
      }
      
      self.SyncState();
      
      if (self.NetworkSync) {
        self.NetworkView.SendMessageAll(self.CreateSyncMessage());
      }
    }
  }
  
  function SyncState() {
    self._internalState = "";
    for (obj in self._targetObjects) {
      if (obj.Active) {
        self._internalState += "1";
      } else {
        self._internalState += "0";
      }
    }
  }
  
  function SyncObject() {
    for (i in Range(0, self._targetObjects.Count, 1)) {
      bit = String.SubstringWithLength(self._internalState, i, 1);
      obj = self._targetObjects.Get(i);
      
      if (bit == "1") {
        obj.Active = true;
      } else {
        obj.Active = false;
      }
    }
  }
  
  function CreateSyncMessage() {
    message = Dict();
    message.Set("internalState", self._internalState);
    message.Set("triggerCount", self._triggerCount);
    return Json.SaveToString(message);
  }
}
