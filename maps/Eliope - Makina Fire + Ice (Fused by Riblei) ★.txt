component Egg
{
    _Touched = false;
    
    CustomEggText = "";
    CustomFontColor = "";
    CustomFontSize = 0;


    ObjectsToShow = "";
    ObjectsToHide = "";


    CustomEggTextTooltip = "Overrides the default EggText of the EggHandler for this specific egg.";
    CustomFontColorTooltip = "Overrides the default EggFontColor for this specific egg.";
    CustomFontSizeTooltip = "Overrides the default EggFontSize for this specific egg.";
    
    ObjectsToShowTooltip = "Enter multiple MapObject IDs separated by space (e.g., 10 11 12) to SHOW when touched. These are hidden on start.";
    ObjectsToHideTooltip = "Enter multiple MapObject IDs separated by space (e.g., 13 14 15) to HIDE when touched.";


    function Init()
    {
        self.ResetStatus();
    }


    function ResetStatus()
    {
        self._Touched = false;
        self.MapObject.Active = true;


        for (id in String.Split(self.ObjectsToShow, " ")) {
            if (id != "") {
                obj = Map.FindMapObjectByID(Convert.ToInt(id));
                if (obj != null) { obj.Active = false; }
            }
        }
        for (id in String.Split(self.ObjectsToHide, " ")) {
            if (id != "") {
                obj = Map.FindMapObjectByID(Convert.ToInt(id));
                if (obj != null) { obj.Active = true; }
            }
        }
    }


    function OnCollisionEnter(other) 
    {
        if (other.Type == "Human" && other.IsMine && !self._Touched)
        {
            self._Touched = true;


            for (id in String.Split(self.ObjectsToShow, " ")) {
                if (id != "") {
                    obj = Map.FindMapObjectByID(Convert.ToInt(id));
                    if (obj != null) { obj.Active = true; }
                }
            }
            for (id in String.Split(self.ObjectsToHide, " ")) {
                if (id != "") {
                    obj = Map.FindMapObjectByID(Convert.ToInt(id));
                    if (obj != null) { obj.Active = false; }
                }
            }


            self.MapObject.Active = false;
        }
    }
}


component EggHandler
{
    Description = "Do not use semicolons (;) to avoid system errors.";


    EggText = "O";
    UseSpace = false;
    EggFontColor = "#FFCC00";
    EggFontSize = "25";
    
    ExtraText = "Eggs to Find∶";
    ExtraFontColor = "White";
    ExtraFontSize = "20";
    ExtraTextPosition = "Above";
    HideExtraTextOnFinish = "Yes";
    
    FinishText = "";
    FinishFontColor = "Red";
    FinishFontSize = "20";


    ObjectsToShow = "";
    ObjectsToHide = "";


    LabelPosition = "MiddleRight";
    PaddingTop = 0;
    PaddingBottom = 0;
    PaddingLeft = 0;    
    PaddingRight = 0;


    ShowServerMessage = true;
    ServerMessage = "has found all the eggs!";


    UseSpaceTooltip = "Add a space between each egg text icon.";
    EggFontColorTooltip = "Use color names or Hex codes like #FFFFFF";
    ExtraTextTooltip = "This text will be displayed near the egg count.";
    ExtraFontColorTooltip = "Use color names or Hex codes like #FFFFFF";
    ExtraTextPositionTooltip = "Choose whether ExtraText appears Above or Below the egg icons.";
    HideExtraTextOnFinishTooltip = "Hide the ExtraText after all eggs are collected.";
    
    FinishTextTooltip = "This message appears only after all eggs are found.";
    FinishFontColorTooltip = "Use color names or Hex codes like #FFFFFF";
    
    ObjectsToShowTooltip = "Enter Object IDs to SHOW after finding all eggs (e.g. 4 5 6). These will be HIDDEN on game start.";
    ObjectsToHideTooltip = "Enter Object IDs to HIDE after finding all eggs (e.g. 7 8 9).";


    LabelPositionTooltip = "Select the UI position on the screen.";
    PaddingTopTooltip = "Push the entire UI down.";
    PaddingBottomTooltip = "Push the entire UI up.";
    PaddingLeftTooltip = "Push the entire UI to the right.";
    PaddingRightTooltip = "Push the entire UI to the left.";


    ExtraTextPositionDropbox = "Above, Below";
    HideExtraTextOnFinishDropbox = "Yes, No";
    LabelPositionDropbox = "TopLeft, TopRight, MiddleLeft, MiddleCenter, MiddleRight, BottomLeft, BottomRight";


    _EggList = List();
    _TotalEggs = 0;
    _Finished = false;


    function Init()
    {
        self._EggList = Map.FindMapObjectsByComponent("Egg");
        self._TotalEggs = self._EggList.Count;
        self.ResetAll();
    }


    function ResetAll()
    {
        self._Finished = false;
        
        for (i in Range(0, self._EggList.Count, 1)) {
            eggObj = self._EggList.Get(i);
            if (eggObj != null) {
                eggComp = eggObj.GetComponent("Egg");
                if (eggComp != null) { eggComp.ResetStatus(); }
            }
        }


        for (id in String.Split(self.ObjectsToShow, " ")) {
            if (id != "") {
                obj = Map.FindMapObjectByID(Convert.ToInt(id));
                if (obj != null) { obj.Active = false; }
            }
        }
        for (id in String.Split(self.ObjectsToHide, " ")) {
            if (id != "") {
                obj = Map.FindMapObjectByID(Convert.ToInt(id));
                if (obj != null) { obj.Active = true; }
            }
        }
    }


    function OnTick()
    {
        currentCount = 0;
        eggIcons = "";
        
        for (i in Range(0, self._EggList.Count, 1))
        {
            eggObj = self._EggList.Get(i);
            if (eggObj != null)
            {
                eggComp = eggObj.GetComponent("Egg");
                if (eggComp != null && !eggComp._Touched)
                {
                    currentCount = currentCount + 1;
                    
                    if (eggComp.CustomEggText != "") { tText = eggComp.CustomEggText; } else { tText = self.EggText; }
                    if (eggComp.CustomFontColor != "") { tColor = eggComp.CustomFontColor; } else { tColor = self.EggFontColor; }
                    if (eggComp.CustomFontSize > 0) { tSize = Convert.ToString(eggComp.CustomFontSize); } else { tSize = self.EggFontSize; }


                    styled = "<size=" + tSize + "><color=" + tColor + ">" + tText + "</color></size>";
                    if (self.UseSpace) { eggIcons = eggIcons + styled + " "; } else { eggIcons = eggIcons + styled; }
                }
            }
        }


        if (self._TotalEggs > 0 && currentCount == 0 && !self._Finished)
        {
            self._Finished = true;


            for (id in String.Split(self.ObjectsToShow, " ")) {
                if (id != "") {
                    obj = Map.FindMapObjectByID(Convert.ToInt(id));
                    if (obj != null) { obj.Active = true; }
                }
            }
            for (id in String.Split(self.ObjectsToHide, " ")) {
                if (id != "") {
                    obj = Map.FindMapObjectByID(Convert.ToInt(id));
                    if (obj != null) { obj.Active = false; }
                }
            }


            if (self.ShowServerMessage)
            {
                playerName = Network.MyPlayer.Name;
                Game.PrintAll(playerName + " " + self.ServerMessage);
            }
        }


        lSpace = ""; for (i in Range(0, self.PaddingLeft, 1)) { lSpace = lSpace + "  "; }
        rSpace = ""; for (i in Range(0, self.PaddingRight, 1)) { rSpace = rSpace + "  "; }
        tSpace = ""; for (i in Range(0, self.PaddingTop, 1)) { tSpace = tSpace + String.Newline; }
        bSpace = ""; for (i in Range(0, self.PaddingBottom, 1)) { bSpace = bSpace + String.Newline; }


        extraStyled = "";
        shouldShowExtra = true;
        if (self._Finished && self.HideExtraTextOnFinish == "Yes") { shouldShowExtra = false; }


        if (self.ExtraText != "" && shouldShowExtra) {
            extraStyled = "<size=" + self.ExtraFontSize + "><color=" + self.ExtraFontColor + ">" + self.ExtraText + "</color></size>";
        }


        displayContent = "";
        if (self.ExtraTextPosition == "Above") {
            if (extraStyled != "") { displayContent = extraStyled + String.Newline; }
            displayContent = displayContent + eggIcons;
        } else {
            displayContent = eggIcons;
            if (extraStyled != "") { displayContent = displayContent + String.Newline + extraStyled; }
        }


        if (self._Finished && self.FinishText != "") {
            displayContent = displayContent + String.Newline + "<size=" + self.FinishFontSize + "><color=" + self.FinishFontColor + ">" + self.FinishText + "</color></size>";
        }


        dot = "<color=#00000000>.</color>";
        finalStr = tSpace + dot + String.Newline + lSpace + displayContent + rSpace + String.Newline + dot + bSpace;


        UI.SetLabel(self.LabelPosition, finalStr);
    }
}


component EggResetter
{
    function OnCollisionEnter(other)
    {
        if (other.Type == "Human" && other.IsMine)
        {
            handlerObjs = Map.FindMapObjectsByComponent("EggHandler");
            if (handlerObjs.Count > 0)
            {
                handler = handlerObjs.Get(0).GetComponent("EggHandler");
                if (handler != null)
                {
                    handler.ResetAll();
                }
            }
        }
    }
}
