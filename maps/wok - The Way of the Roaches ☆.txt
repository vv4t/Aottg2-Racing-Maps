component FengGameManagerMKII {
  _captured = 0;

  function OnGameStart() {
    self._goals = Map.FindMapObjectsByComponent("RacingGoal");
    self._finish = Map.FindMapObjectByComponent("RacingFinishRegion");
    self.OnResetRace();
    
    if (RoomData.GetProperty("StartAlert", true)) {
      text = "Capture all the goals in any order, then head to the finish.\nYou can stock or run through the course.";
      Game.Print("<color=green>" + text + "</color>");
      RoomData.SetProperty("StartAlert", false);
    }
  }

  function OnCharacterSpawn(c) {
    if (c.Type == "Human" && c.IsMine) {
      c.MaxSpeed = 300;
      c.Speed = 300;
    }
  }

  function OnSecond() {
    me = Network.MyPlayer.Character;
    if (me != null) {
      me.SetSpecial(SpecialEnum.Potato);
    }
  }

  function OnResetRace() {
    self._captured = 0;
    for (goal in self._goals) {
      goal.Active = true;
    }
    self._finish.Active = false;
  }

  function CaptureGoal() {
    self._captured += 1;
    UI.SetLabelForTime("MiddleCenter", "Captured <color=#aaffaa>" + self._captured + "/" + self._goals.Count + "</color>.", 1.0);

    if (self._captured == self._goals.Count) {
      self._finish.Active  = true;
      self.EndAlert();
    }
  }

  coroutine EndAlert() {
    wait 1.0;
    UI.SetLabelForTime("MiddleCenter", "<color=#FFBB00>Head to the finish.</color>", 3.0);
  }
}

component RacingGoal {
  function OnGameStart() {
    self.g = Map.FindMapObjectByName("FengGameManagerMKII").GetComponent("FengGameManagerMKII");
  }

  function OnCollisionEnter(c) {
    if (c.Type == "Human" && c.IsMine) {
      self.g.CaptureGoal();
      self.MapObject.Active = false;
    }
  }
}
